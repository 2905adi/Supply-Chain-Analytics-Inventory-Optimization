df["Forecast_Error"] = (
    df["Units_Sold"] - df["Demand_Forecast"]
)

df["Absolute_Error"] = df["Forecast_Error"].abs()

forecast_accuracy = (
    df.groupby("SKU_ID")
      .agg(
          MAE=("Absolute_Error", "mean"),
          Bias=("Forecast_Error", "mean"),

    forecast_accuracy["MAE"] /

planning_risk = demand_volatility[
    ["SKU_ID", "CV"]
].merge(
    forecast_accuracy[
        ["SKU_ID", "MAE_%"]
    ],
    on="SKU_ID"
)

cv_threshold = planning_risk["CV"].median()
mae_threshold = planning_risk["MAE_%"].median()

planning_risk["Planning_Profile"] = np.select(
    [
        (planning_risk["CV"] > cv_threshold) &
        (planning_risk["MAE_%"] > mae_threshold),

        (planning_risk["CV"] > cv_threshold) |
        (planning_risk["MAE_%"] > mae_threshold)
    ],
    [
        "High Uncertainty",
        "Moderate Uncertainty"
    ],
    default="Stable"
)

planning_risk = planning_risk.sort_values(
    ["Planning_Profile", "CV"],
    ascending=[True, False]
)

planning_risk.round(2)

optimization = (
    df.groupby(["SKU_ID", "Warehouse_ID"])
    .agg(
        Avg_Demand=("Units_Sold", "mean"),
        Demand_Std=("Units_Sold", "std"),
        Avg_Lead_Time=("Supplier_Lead_Time_Days", "mean"),
        Current_Inventory=("Inventory_Level", "mean"),
        Current_ROP=("Reorder_Point", "mean"),
        Avg_Order_Qty=("Order_Quantity", "mean"),
        Unit_Cost=("Unit_Cost", "mean")
    )
    .reset_index()
)

optimization["Lead_Time_Demand"] = (
    optimization["Avg_Demand"] *
    optimization["Avg_Lead_Time"]
)

optimization["Safety_Stock"] = (
    Z *
    optimization["Demand_Std"] *
    np.sqrt(optimization["Avg_Lead_Time"])
)

optimization["Demand_Driven_ROP"] = (
    optimization["Lead_Time_Demand"] +
    optimization["Safety_Stock"]
).round()
